# Agent API Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Extend the `agentshield_hook_events` stream with `deployment_id`/`deployment_name`, stamp those fields from the plugin config, and replace the mock `GET /api/agents` and `GET /api/agents/:id` handlers with real Proton queries.

**Architecture:** Direct-query — each API request POSTs a `GROUP BY` SQL query to the Proton HTTP endpoint (`POST /`). A new thin `timeplus.Query()` helper handles auth and response parsing. No caching, no backward-compat shims for the old stream schema (operator drops the stream manually once before deploying).

**Tech Stack:** TypeScript (OpenClaw plugin), Go/Gin (backend), Timeplus Proton HTTP API (`FORMAT JSON`), Python (test runner).

- [ ] **Step 1: Create `backend/internal/timeplus/client.go`**

```go
package timeplus

import (
	"encoding/json"
	"fmt"
	"io"
	"net/http"
	"strings"

	"github.com/timeplus/agentshield/internal/config"
)

// Row is a single result row from a Proton query.
type Row = map[string]any

// Query executes a SELECT SQL statement against the Proton HTTP interface and
// returns all rows. It appends "FORMAT JSON" to the query automatically.
// Returns a wrapped error on non-2xx status or parse failure.
func Query(cfg *config.TimeplusConfig, sql string) ([]Row, error) {
	url := fmt.Sprintf("http://%s:%d/", cfg.Host, cfg.Port)
	body := strings.NewReader(sql + " FORMAT JSON")

	req, err := http.NewRequest(http.MethodPost, url, body)
	if err != nil {
		return nil, fmt.Errorf("proton query: build request: %w", err)
	}
	req.Header.Set("Content-Type", "text/plain")
	if cfg.User != "" {
		req.SetBasicAuth(cfg.User, cfg.Password)
	}

	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		return nil, fmt.Errorf("proton query: %w", err)
	}
	defer resp.Body.Close()

	if resp.StatusCode != http.StatusOK {
		errBody, _ := io.ReadAll(resp.Body)
		return nil, fmt.Errorf("proton query: status %d: %s", resp.StatusCode, string(errBody))
	}

	var result struct {
		Data []Row `json:"data"`
	}
	if err := json.NewDecoder(resp.Body).Decode(&result); err != nil {
		return nil, fmt.Errorf("proton query: decode: %w", err)
	}
	return result.Data, nil
}
```

```go
package handlers

import (
	"fmt"
	"net/http"
	"strconv"
	"strings"
	"time"

	"github.com/gin-gonic/gin"
	"github.com/timeplus/agentshield/internal/config"
	"github.com/timeplus/agentshield/internal/models"
	"github.com/timeplus/agentshield/internal/timeplus"
)

func ListAgents(cfg *config.Config) gin.HandlerFunc {
	return func(c *gin.Context) {
		pageStr := c.DefaultQuery("page", "1")
		limitStr := c.DefaultQuery("limit", "15")

		page, err := strconv.Atoi(pageStr)
		if err != nil || page < 1 {
			page = 1
		}
		limit, err := strconv.Atoi(limitStr)
		if err != nil || limit < 1 {
			limit = 15
		}

		sql := `
SELECT
    agent_id,
    deployment_id,
    deployment_name,
    max(event_time)             AS last_seen,
    count()                     AS event_count,
    count(DISTINCT session_key) AS session_count
FROM table(agentshield_hook_events)
GROUP BY agent_id, deployment_id, deployment_name
ORDER BY last_seen DESC`

		rows, err := timeplus.Query(&cfg.Timeplus, sql)
		if err != nil {
			c.JSON(http.StatusServiceUnavailable, gin.H{"error": "data store unavailable"})
			return
		}

		agents := make([]models.Agent, 0, len(rows))
		for _, row := range rows {
			a, parseErr := rowToAgent(row)
			if parseErr != nil {
				continue // skip malformed rows
			}
			agents = append(agents, a)
		}

		total := len(agents)
		start := (page - 1) * limit
		if start >= total {
			c.JSON(http.StatusOK, models.AgentListResponse{
				Agents: []models.Agent{},
				Total:  total,
				Page:   page,
				Limit:  limit,
			})
			return
		}
		end := start + limit
		if end > total {
			end = total
		}

		c.JSON(http.StatusOK, models.AgentListResponse{
			Agents: agents[start:end],
			Total:  total,
			Page:   page,
			Limit:  limit,
		})
	}
}

func GetAgent(cfg *config.Config) gin.HandlerFunc {
	return func(c *gin.Context) {
		id := c.Param("id")
		parts := strings.SplitN(id, "__", 2)
		if len(parts) != 2 {
			c.JSON(http.StatusBadRequest, gin.H{"error": "invalid agent id format, expected deploymentId__agentId"})
			return
		}
		deploymentID := parts[0]
		agentID := parts[1]

		sql := fmt.Sprintf(`
SELECT
    agent_id,
    deployment_id,
    deployment_name,
    max(event_time)             AS last_seen,
    count()                     AS event_count,
    count(DISTINCT session_key) AS session_count
FROM table(agentshield_hook_events)
WHERE deployment_id = '%s'
  AND agent_id      = '%s'
GROUP BY agent_id, deployment_id, deployment_name
LIMIT 1`,
			deploymentID, agentID)

		rows, err := timeplus.Query(&cfg.Timeplus, sql)
		if err != nil {
			c.JSON(http.StatusServiceUnavailable, gin.H{"error": "data store unavailable"})
			return
		}
		if len(rows) == 0 {
			c.JSON(http.StatusNotFound, gin.H{"error": "not found"})
			return
		}

		a, err := rowToAgent(rows[0])
		if err != nil {
			c.JSON(http.StatusInternalServerError, gin.H{"error": "failed to parse agent data"})
			return
		}
		c.JSON(http.StatusOK, a)
	}
}

func rowToAgent(row map[string]any) (models.Agent, error) {
	agentID, _ := row["agent_id"].(string)
	deploymentID, _ := row["deployment_id"].(string)
	deploymentName, _ := row["deployment_name"].(string)

	var lastSeen time.Time
	if ls, ok := row["last_seen"].(string); ok && ls != "" {
		t, err := time.Parse("2006-01-02 15:04:05.999999999 -0700 MST", ls)
		if err != nil {
			// Proton returns various time formats; try RFC3339 fallback.
			t, err = time.Parse(time.RFC3339Nano, ls)
			if err != nil {
				lastSeen = time.Time{}
			} else {
				lastSeen = t
			}
		} else {
			lastSeen = t
		}
	}

	eventCount := 0
	switch v := row["event_count"].(type) {
	case float64:
		eventCount = int(v)
	case int:
		eventCount = v
	}

	sessionCount := 0
	switch v := row["session_count"].(type) {
	case float64:
		sessionCount = int(v)
	case int:
		sessionCount = v
	}

	return models.Agent{
		AgentID:        agentID,
		DeploymentID:   deploymentID,
		DeploymentName: deploymentName,
		LastSeen:       lastSeen,
		EventCount:     eventCount,
		SessionCount:   sessionCount,
	}, nil
}
```

Add a helper `_load_proton_config()` next to `_load_openclaw_token()`:

```python
def _load_proton_config() -> tuple[str, str, str]:
    """Read Proton URL, username, and password from openclaw.json plugin config."""
    config_path = Path(__file__).parent.parent / ".openclaw" / "openclaw.json"
    with config_path.open() as f:
        data = json.load(f)
    plugin_cfg = data["plugins"]["entries"]["agentshield"]["config"]
    url = plugin_cfg.get("timeplusUrl", "http://localhost:3218")
    # timeplusUrl uses host.docker.internal inside Docker — remap to localhost for host-side Python
    url = url.replace("host.docker.internal", "localhost")
    user = plugin_cfg.get("username", "proton")
    password = plugin_cfg.get("password", "")
    return url, user, password
```

**Separator:** `__` (double underscore) used in `GetAgent` `:id` path parameter — correct, avoids collision with colon-containing `agent_id` values like `agent:main`.

**`DashboardResponse.RecentAgents`:** `mock.Agents` is now `[]models.Agent{}` (empty). The dashboard endpoint continues to compile and return an empty `recent_agents` array — acceptable per spec (frontend changes are out of scope).

**SQL injection note:** `GetAgent` interpolates `deploymentID` and `agentID` directly into the SQL string. These values come from the URL path parameter, which is attacker-controlled. The risk is mitigated by the `__` split requiring exactly two parts, but a follow-up should use parameterized queries once Proton's Go driver supports them via HTTP. This is noted as a known limitation, not a blocker.
